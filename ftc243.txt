    <script>
        // Data Handling with LocalStorage
        let team = JSON.parse(localStorage.getItem('robotics_db_team')) || [];
        let finance = JSON.parse(localStorage.getItem('robotics_db_finance')) || [];

        function save() {
            localStorage.setItem('robotics_db_team', JSON.stringify(team));
            localStorage.setItem('robotics_db_finance', JSON.stringify(finance));
            render();
        }

        function showTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
            document.querySelectorAll('.nav-link').forEach(l => l.classList.remove('active'));
            document.getElementById(tabId).classList.add('active');
            document.getElementById('btn-' + tabId).classList.add('active');
        }

        function addMember() {
            const name = document.getElementById('tName').value;
            const phone = document.getElementById('tPhone').value;
            const role = document.getElementById('tRole').value;
            if(!name || !phone) return alert("אנא הזן את כל הפרטים");
            team.push({ name, phone, role });
            document.getElementById('tName').value = ''; 
            document.getElementById('tPhone').value = '';
            save();
        }

        function deleteMember(index) {
            if(confirm("למחוק חבר צוות זה?")) {
                team.splice(index, 1);
                save();
            }
        }

        function addTransaction() {
            const amt = parseFloat(document.getElementById('fAmount').value);
            const type = document.getElementById('fType').value;
            const desc = document.getElementById('fDesc').value;
            if(!amt || !desc) return alert("אנא הזן סכום ותיאור");
            const date = new Date().toLocaleDateString('he-IL');
            finance.push({ date, desc, type, amt });
            document.getElementById('fAmount').value = '';
            document.getElementById('fDesc').value = '';
            save();
        }

        function deleteFinance(index) {
            if(confirm("למחוק תנועה זו?")) {
                finance.splice(index, 1);
                save();
            }
        }

        function render() {
            // Render Team Table
            document.getElementById('teamBody').innerHTML = team.map((m, i) => `
                <tr class="border-b">
                    <td class="p-3 font-bold">${m.name}</td>
                    <td class="p-3 text-indigo-600">${m.role}</td>
                    <td class="p-3">${m.phone}</td>
                    <td class="p-3"><button onclick="deleteMember(${i})" class="text-red-500 font-bold">מחק</button></td>
                </tr>`).join('');

            // Render Selection List (WhatsApp)
            document.getElementById('selectionList').innerHTML = team.length ? team.map((m, i) => `
                <div class="flex items-center justify-between p-3 border-b hover:bg-indigo-50 transition">
                    <div class="flex items-center gap-3">
                        <input type="checkbox" id="check-${i}" class="w-5 h-5 accent-indigo-600">
                        <label for="check-${i}" class="font-bold cursor-pointer text-slate-700">${m.name}</label>
                    </div>
                    <span class="text-xs text-gray-400 italic">${m.role}</span>
                </div>`).join('') : '<p class="text-center text-gray-400 py-4">אין חברים ברשימה</p>';

            // Render Finance
            let inc = 0, exp = 0;
            document.getElementById('financeBody').innerHTML = finance.map((f, i) => {
                if(f.type === 'plus') inc += f.amt; else exp += f.amt;
                return `
                <tr class="border-b hover:bg-gray-50">
                    <td class="p-3 text-xs text-gray-400">${f.date}</td>
                    <td class="p-3 font-bold">${f.desc}</td>
                    <td class="p-3 font-bold ${f.type === 'plus' ? 'text-green-600' : 'text-red-600'}">${f.type === 'plus' ? 'הכנסה (+)' : 'הוצאה (-)'}</td>
                    <td class="p-3 font-bold">₪${f.amt}</td>
                    <td class="p-3"><button onclick="deleteFinance(${i})" class="text-red-500">מחק</button></td>
                </tr>`;
            }).join('');

            document.getElementById('totalInc').innerText = `₪${inc}`;
            document.getElementById('totalExp').innerText = `₪${exp}`;
            document.getElementById('balanceVal').innerText = `₪${inc - exp}`;
        }

        function sendBulkWhatsApp() {
            const title = document.getElementById('mTitle').value;
            const date = document.getElementById('mDate').value;
            if(!title || !date) return alert("נא להזין נושא ותאריך פגישה");

            team.forEach((m, i) => {
                if(document.getElementById(`check-${i}`).checked) {
                    const msg = `שלום ${m.name},%0A%0Aהודעה מצוות הרובוטיקה:%0Aנקבעה פגישה בנושא: *${title}*%0Aבתאריך ושעה: *${date.replace('T', ' ')}*%0Aתפקידך: *${m.role}*%0A%0Aנא לאשר הגעה ולבוא בזמן! `;
                    window.open(`https://wa.me/${m.phone}?text=${msg}`, '_blank');
                }
            });
        }

        window.onload = render;
    </script>
